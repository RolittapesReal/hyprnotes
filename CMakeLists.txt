cmake_minimum_required(VERSION 3.20)
project(hyprnotes VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Sql Svg Test)
enable_testing()
# The plugins Hyprnotes maintains live in their own repository next to this one; the plugin tests read them from here
# and skip themselves when the folder is absent.
set(HN_PLUGINS_DIR "${CMAKE_SOURCE_DIR}/../hyprnotes-plugins" CACHE PATH "folder of the hyprnotes-plugins repository (used by tests only)")
set(HN_SANITIZE "" CACHE STRING "sanitizers for every target, e.g. address,undefined or thread")
option(HN_COVERAGE "build with gcov instrumentation (use Debug)" OFF)
option(HN_FUZZ "build libFuzzer targets in tests/fuzz (Clang only)" OFF)
option(HN_LIVE_TESTS "build test-only Hyprland live-test driver" OFF)
if(HN_SANITIZE)
  add_compile_options(-fsanitize=${HN_SANITIZE} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${HN_SANITIZE})
endif()
if(HN_COVERAGE)
  add_compile_options(--coverage -O0 -g)
  add_link_options(--coverage)
endif()
if(HN_FUZZ)
  add_compile_options(-fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer -g -O1)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
# Each module owns its own CMakeLists.txt and defines a static library hn_<module>.
set(HN_MODULES core;theme;editor;platform;mods;plugins;app CACHE STRING "modules to build (for parallel development)")
foreach(mod ${HN_MODULES})
  if(EXISTS ${CMAKE_SOURCE_DIR}/src/${mod}/CMakeLists.txt)
    add_subdirectory(src/${mod})
  endif()
endforeach()

add_subdirectory(tests/fuzz)
if(HN_LIVE_TESTS)
  add_subdirectory(tests/live)
endif()

include(cmake/HyprnotesInstall.cmake)
if(HN_STRESS)
  add_subdirectory(tests/stress)
endif()
